vec4 getResiduals(sampler2D tex, vec4 which_channel, vec2 uv, vec2 tex_size_pixels) {
    vec2 inv_res = 1.0 / tex_size_pixels;
    int half_k = 1;

    const float conv_w[72] = float[72](
        -0.09768, -0.18516, 0.00000, -0.33358, 1.51569, -0.26869, -0.10608, -0.22757, -0.30178, -0.82323, 0.10377, 0.04784, -0.05184, 1.12308, -0.35418, 0.06491,
        -0.19375, 0.11747, 0.03427, 0.04078, -0.07082, 0.05485, -1.11042, 0.05724, 0.06996, 0.89836, 0.03078, 0.01727, 0.00000, 0.00000, -0.05531, -1.01892,
        1.09752, 0.00000, -0.06983, 0.02342, 0.14619, -0.28688, -0.08622, -0.16032, 1.55682, -0.42973, 0.01786, -0.09117, -0.66628, 0.28674, -0.59189, -0.38449,
        -0.38299, 1.09850, -0.03752, 0.00000, 0.14109, -0.14607, 0.22264, 0.75134, 0.00000, 0.12046, -1.18854, 0.03827, 0.08882, 0.00000, -0.03374, 0.13421,
        0.49176, 0.01726, 0.08578, -1.47278, 0.09623, 0.05584, 0.56471, 0.02834
    );
    const float conv_b[8] = float[8](
        0.00000, -0.04260, 0.00000, -0.02525, -0.02869, 0.01211, 0.00000, 0.00000
    );
    const float mid_0_w[48] = float[48](
        -0.13783, 0.09662, -0.54473, 0.79772, 0.25340, 0.22585, -0.50966, 0.85103, 0.50286, 1.45360, 0.03280, -0.01846, -0.40413, 0.20832, 0.48736, 0.47405,
        0.67251, -0.14465, 0.70356, 0.02437, -0.32057, 0.08009, -2.42391, -0.86720, -1.13758, -0.70537, -0.54275, 0.34273, 0.64294, -0.03248, -0.52081, 0.89544,
        0.76423, 1.80197, -0.31926, -0.04082, -0.32862, 0.38305, 0.37048, 0.53183, 0.58770, 0.78918, -0.76948, -0.01465, -0.30647, 0.47474, 0.69968, -0.98733
    );
    const float mid_0_b[6] = float[6](
        0.00000, 0.00000, -0.01249, 0.00000, 0.00000, -0.01087
    );
    const float zfinal_w[24] = float[24](
        -0.47789, 0.05468, -0.28100, 1.05227, -0.34796, 0.84717, 0.54367, 0.21103, -0.45967, -1.41424, -0.40584, 0.51105, -0.57291, -0.58944, 0.79771, 1.28245,
        0.62077, -0.31418, 0.43941, -0.51467, 0.35454, -1.19178, 0.73684, -0.79637
    );
    const float zfinal_b[4] = float[4](
        0.00000, 0.00000, 0.00000, 0.00000
    );

    float neigh[9];
    int idx = 0;
    for (int dy = -half_k; dy <= half_k; dy++) {
        for (int dx = -half_k; dx <= half_k; dx++) {
            vec2 offset = vec2(float(dx), float(dy)) * inv_res;
            neigh[idx++] = dot(texture2DLod(tex, uv + offset, 0.0), which_channel);
        }
    }

    float h0[8];
    int w_idx = 0;
    for (int i=0; i<8; i++) {
        h0[i] = conv_b[i];
        for (int j=0; j<9; j++) {
            h0[i] += neigh[j] * conv_w[w_idx++];
        }
        h0[i] = max(0.0, h0[i]);
    }

    float h1[6];
    w_idx = 0;
    for (int i=0; i<6; i++) {
        h1[i] = mid_0_b[i];
        for (int j=0; j<8; j++) {
            h1[i] += h0[j] * mid_0_w[w_idx++];
        }
        h1[i] = max(0.0, h1[i]);
    }

    vec4 out_val = vec4(zfinal_b[0], zfinal_b[1], zfinal_b[2], zfinal_b[3]);
    w_idx = 0;
    for (int i=0; i<4; i++) {
        for (int j=0; j<6; j++) {
            out_val[i] += h1[j] * zfinal_w[w_idx++];
        }
    }

    return out_val;
}
